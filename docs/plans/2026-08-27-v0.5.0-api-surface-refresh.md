# v0.5.0 API Surface Refresh Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bring the plugin to parity with the Descript API as audited on 2026-08-27 - two new commands (`descript models`, `descript transcript`), refreshed spec baseline and types, `publishes` surfaced in `projects get`, `--workspace` on import, matching MCP tools and skills.

**Architecture:** The CLI stays the single source of truth (registry command handlers over a thin typed client), the MCP shim only maps argv, skills only document. The one structural addition is a raw-body path on `HttpClient` because `POST /export/transcript` returns a file, not JSON. Everything else is additive fields and two new client functions.

**Tech Stack:** TypeScript strict ESM (imports carry `.js` suffixes), zero runtime dependencies, Node >= 24 built-in test runner over compiled output.

**Source of truth for all schemas:** the live OpenAPI extracted in the 2026-08-27 audit (see `docs/field-reports/2026-08-27-api-drift-audit.md`). Task 1 lands it at `docs/descript-openapi.json`.

## Global Constraints

- Node >= 24, `"type": "module"`, no runtime dependencies (devDeps only - typescript, @types/node).
- Tests run via `npm test` which is `tsc -p tsconfig.json && node --test "dist/tests/**/*.test.js"`. Never use bare `node --test` without the dist glob (broken on this machine's Node 24 setup).
- All work on branch `v0.5.0`, one commit per task. Push, merge to main, tag, and plugin cache refresh happen only at session close per change-control - never mid-plan.
- The MCP shim must re-use `runCli`; new tools are `TOOLS` entries plus argv builders only.
- New skills classify against `docs/specs/2026-05-20-model-invocation-policy.md` - both new commands are read-only and free, therefore unrestricted and model-invocable.
- Formatting in docs and skills - no em or en dashes, no colons in headings (use " - "), straight quotes only.
- Types are compile-time trust, not runtime validation - new response fields land as optional properties so unexpected vendor payloads never throw.
- Out of scope (stays in `docs/plans/2026-05-21-v0.5.0-backlog.md` for later releases) - Theme 1 Underlord skill expansion, Theme 2 test hardening, Theme 3 resume generalization, Theme 5 distribution, Theme 7 cosmetics, and the `export/timeline` job type (dangling ref in the vendor spec, no public creation path - do not build).

```ts
  // Same auth, query, retry, and error mapping as request(), but returns the
  // response body as bytes for endpoints that serve files (POST /export/transcript
  // returns txt/markdown/html/rtf/srt as text and docx as binary).
  async requestRaw(method: string, path: string, opts: RequestOptions = {}): Promise<RawResponse> {
    const url = new URL(this.baseUrl + path);
    for (const [k, v] of Object.entries(opts.query ?? {})) {
      if (v !== undefined) url.searchParams.set(k, String(v));
    }
    const headers: Record<string, string> = {
      ...(opts.headers ?? {}),
      authorization: `Bearer ${this.token}`,
      accept: "*/*"
    };
    let init: RequestInit = { method, headers };
    if (opts.body !== undefined) {
      headers["content-type"] = "application/json";
      init = { ...init, body: JSON.stringify(opts.body) };
    }
    let attempt = 0;
    for (;;) {
      const resp = await fetch(url.toString(), init);
      if (resp.status === 429 && attempt < this.maxRetries) {
        const wait = retryAfterMs(resp);
        attempt += 1;
        await this.sleep(wait);
        continue;
      }
      if (resp.status === 204) return { bytes: new Uint8Array(0) };
      if (resp.ok) {
        return {
          bytes: new Uint8Array(await resp.arrayBuffer()),
          contentType: resp.headers.get("content-type") ?? undefined
        };
      }
      throw await toApiError(resp);
    }
  }
```

```ts
import type { StatusResponse } from "../../client/types.js";

// Any 2xx from /status means the token authenticated (an invalid token throws
// 401 before this formatter runs). The endpoint stabilized in the 2026-08-27
// spec refresh - { drive_id, drive_name, api_version } - but every field stays
// optional so a degraded payload never interpolates a bare JS "undefined".
export function formatStatus(r: StatusResponse | undefined): string {
  const parts: string[] = [];
  if (r?.drive_name) parts.push(`drive ${r.drive_name}`);
  if (r?.drive_id) parts.push(`drive_id ${r.drive_id}`);
  if (r?.api_version) parts.push(`API ${r.api_version}`);
  const detail = parts.length ? ` (${parts.join(", ")})` : "";
  return `Authenticated to Descript${detail}.`;
}
```

```ts
// Collects the four --timecodes-* flags into the API's timecodes object.
// Returns undefined when no flag was passed, null after emitting a usage error.
function buildTimecodes(ctx: Ctx): TranscriptTimecodeOptions | undefined | null {
  const every = ctx.flags["timecodes-every"];
  const offset = ctx.flags["timecodes-offset"];
  const onParagraphs = ctx.flags["timecodes-on-paragraphs"] === true;
  const onMarkers = ctx.flags["timecodes-on-markers"] === true;
  if (every === undefined && offset === undefined && !onParagraphs && !onMarkers) return undefined;
  const t: TranscriptTimecodeOptions = {};
  if (every !== undefined) {
    const n = Number(every);
    if (!Number.isFinite(n) || n <= 0) { fail(ctx.io, "--timecodes-every must be a positive number of seconds"); return null; }
    t.frequency_seconds = n;
  }
  if (offset !== undefined) {
    const n = Number(offset);
    if (!Number.isFinite(n)) { fail(ctx.io, "--timecodes-offset must be a number of seconds"); return null; }
    t.offset_seconds = n;
  }
  if (onParagraphs) t.on_paragraphs = true;
  if (onMarkers) t.on_markers = true;
  return t;
}
```

```ts
  async transcript(ctx) {
    const projectId = ctx.args[0];
    if (!projectId) {
      fail(ctx.io, "Usage: descript transcript <project-id> [composition-id] --format txt|markdown|html|rtf|docx|srt [--out <path>]");
      return 2;
    }
    if (badEnum(ctx, "format", TRANSCRIPT_FORMAT)) return 2;
    if (badEnum(ctx, "speaker-labels", SPEAKER_LABELS)) return 2;
    const format = ctx.flags.format;
    if (typeof format !== "string") {
      fail(ctx.io, "--format is required (txt|markdown|html|rtf|docx|srt)");
      return 2;
    }
    const out = typeof ctx.flags.out === "string" ? ctx.flags.out : undefined;
    if (format === "docx" && !out) {
      fail(ctx.io, "--out <path> is required for docx (binary output cannot go to the terminal)");
      return 2;
    }
    const timecodes = buildTimecodes(ctx);
    if (timecodes === null) return 2;
    const req: TranscriptExportRequest = {
      project_id: projectId,
      ...(ctx.args[1] ? { composition_id: ctx.args[1] } : {}),
      format: format as TranscriptFormat,
      ...(typeof ctx.flags["speaker-labels"] === "string"
        ? { include_speaker_labels: ctx.flags["speaker-labels"] as "off" | "changes" | "every_paragraph" } : {}),
      ...(ctx.flags.markers === true ? { include_markers: true } : {}),
      ...(timecodes ? { timecodes } : {})
    };
    const r = await client(ctx).exportTranscript(req);
    if (out) {
      writeFileSync(out, r.bytes);
      emit(ctx.io, `Wrote ${out} (${r.bytes.length} bytes, ${format})`, { ok: true, path: out, bytes: r.bytes.length, format });
      return 0;
    }
    const text = new TextDecoder().decode(r.bytes);
    if (ctx.io.json) {
      emit(ctx.io, text, { format, content: text });
    } else {
      ctx.io.stdout(text.endsWith("\n") ? text : text + "\n");
    }
    return 0;
  },
```
